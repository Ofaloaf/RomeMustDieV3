### Culture name assignments
# Italia Region
assign_state_name_region_italy  = {
	if = {
        limit = {
            state_region = {
                s:STATE_VENETIA ?= this
            }
        }
        STATE_VENETIA_state_name_assign = yes
    }	
}

# Illyricum Region
assign_state_name_region_balkans  = {
    if = {
        limit = {
            state_region = {
                s:STATE_DALMATIA ?= this
            }
        }
        STATE_DALMATIA_state_name_assign = yes
    }   
    if = {
        limit = {
            state_region = {
                s:STATE_ALBANIA ?= this
            }
        }
        STATE_ALBANIA_state_name_assign = yes
    }   
}

# Dnieper Region

STATE_VENETIA_state_name_assign = {
	if = {
        limit = {
            owner = {
                any_primary_culture = {
                    cu:greek ?= this
                }
				NOT = {
					any_primary_culture = {
						cu:roman ?= this
					}
				}
            }
        }
		
        set_state_name = STATE_VENETIA_greek
        
        set_hub_names = greek
    }
	else = { # By putting an else at the end that resets the names we ensure that the names are switched back to their defaults if anyone else takes the states
		reset_state_name = yes
		reset_hub_names = yes
	}
}

STATE_ALBANIA_state_name_assign = {
    if = {
        limit = {
            owner = {
                any_primary_culture = {
                    cu:greek ?= this
                }
                NOT = {
                    any_primary_culture = {
                        cu:roman ?= this
                    }
                }
            }
        }
        
        set_state_name = STATE_ALBANIA_greek
        
        set_hub_names = greek
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = {
                    has_discrimination_trait = latin_speaker
                }
                NOT = {
                    owner = {
                        any_primary_culture = { cu:roman = this }
                        coa_republic_trigger = yes
                    }
                }
            }
        }
        
        set_state_name = STATE_ALBANIA_latin
        
        set_hub_names = latin
    }
    else_if = {
        limit = {
            owner = {
                any_primary_culture = { cu:roman = this }
                coa_republic_trigger = yes
            }
        }
        
        set_state_name = STATE_ALBANIA_latin
        
        set_hub_name = {
            type = city
            name = HUB_NAME_STATE_ALBANIA_city_vulgar_latin
        }

        set_hub_name = {
            type = port
            name = HUB_NAME_STATE_ALBANIA_port_vulgar_latin
        }

        set_hub_name = {
            type = mine
            name = HUB_NAME_STATE_ALBANIA_mine_latin
        }
        
        set_hub_name = {
            type = farm
            name = HUB_NAME_STATE_ALBANIA_farm_vulgar_latin
        }
        
        set_hub_name = {
            type = wood
            name = HUB_NAME_STATE_ALBANIA_wood_vulgar_latin
        }
    }
    else = { # By putting an else at the end that resets the names we ensure that the names are switched back to their defaults if anyone else takes the states
        reset_state_name = yes
        reset_hub_names = yes
    }
}
